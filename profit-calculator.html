function calc(){ var price=parseFloat(document.getElementById("price").value)||0; var cost=parseFloat(document.getElementById("cost").value)||0; var shipping=parseFloat(document.getElementById("shipping").value)||0; var ads=parseFloat(document.getElementById("ads").value)||0; var fee=parseFloat(document.getElementById("fee").value)||0; var payment=price*(fee/100); var profit=price-cost-shipping-ads-payment; document.getElementById("result").innerHTML="Profit: $"+profit.toFixed(2); }
